# ForegramPM — message catalogue, English (en)
#
# One entry per line: `key = template`. In a template, `{name}` is replaced
# by the argument of that name; any other brace is kept as is; `\n` starts
# a new line.
# A refusal key starts with its stable code (GPH003), followed if needed by a
# variant (JSN005.zero). The other keys are fragments that refusals embed.

# JSON syntax
JSN001.utf8 = invalid bytes: the text must be UTF-8
JSN001.bom = byte order mark (BOM) refused at the start of the text
JSN002.value-char = unexpected character `{char}`: a value is expected
JSN002.value-word = unknown word `{word}`: a value is expected
JSN002.expected = unexpected character `{char}`: {expected} expected
JSN003.value = unexpected end of text: a value is expected
JSN003.expected = unexpected end of text: {expected} expected
JSN003.string = unterminated string
JSN004 = non-integer number: only integers are allowed
JSN005.digit = malformed number: a digit is expected after `-`
JSN005.zero = malformed number: leading zero
JSN005.minus-zero = malformed number: `-0` is written `0`
JSN006 = integer too large to be represented
JSN007.control = unescaped control character {char} in a string
JSN007.high = high surrogate without its low surrogate
JSN007.low = lone low surrogate
JSN007.escape = unknown escape sequence
JSN007.hex = four hexadecimal digits are expected after `\u`
JSN008 = duplicate key `{key}` in the same object
JSN009 = nesting too deep: at most {max} levels
JSN010 = unexpected content after the value: `{char}`

# What the JSON reader expected
expect.comma-bracket = `,` or `]`
expect.comma-brace = `,` or `}`
expect.colon = `:`
expect.key = a quoted key
