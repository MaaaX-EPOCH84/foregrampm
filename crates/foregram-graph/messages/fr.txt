# ForegramPM — catalogue de messages, français (fr)
#
# Une entrée par ligne : `clé = modèle`, comme dans en.txt, dont ce fichier
# reprend les clés du produit (le banc, ses familles bench et board et ses
# refus X, E, M et R, ne parle qu'anglais) avec les mêmes emplacements.
# Typographie : guillemets « » autour des noms, accents graves autour des
# clés du format, espace insécable (U+00A0) avant : ; ! ? et dans les « ».
# Un repère de fichier ou de position (« g.json:1:1: ») garde la forme
# anglaise, que les éditeurs savent lire.
# Pluriel : la variante `.one` sert pour 0 et 1, `.other` au-delà.

# Syntaxe JSON
J001.utf8 = octets invalides : le texte doit être en UTF-8
J001.bom = indicateur d'ordre des octets (BOM) refusé au début du texte
J002.value-char = caractère inattendu `{char}` : on attend une valeur
J002.value-word = mot inconnu `{word}` : on attend une valeur
J002.expected = caractère inattendu `{char}` : on attend {expected}
J003.value = fin de texte inattendue : on attend une valeur
J003.expected = fin de texte inattendue : on attend {expected}
J003.string = chaîne non terminée
J004 = nombre non entier : seuls les entiers sont admis
J005.digit = nombre mal formé : un chiffre est attendu après `-`
J005.zero = nombre mal formé : zéro en tête
J005.minus-zero = nombre mal formé : `-0` s'écrit `0`
J006 = entier trop grand pour être représenté
J007.control = caractère de contrôle {char} non échappé dans une chaîne
J007.high = surrogate haut sans son surrogate bas
J007.low = surrogate bas isolé
J007.escape = séquence d'échappement inconnue
J007.hex = quatre chiffres hexadécimaux sont attendus après `\u`
J008 = clé `{key}` en double dans le même objet
J009 = imbrication trop profonde : {max} niveaux au plus
J010 = contenu inattendu après la valeur : `{char}`

# Ce qu'attendait le lecteur JSON
expect.comma-bracket = `,` ou `]`
expect.comma-brace = `,` ou `}`
expect.colon = `:`
expect.key = une clé entre guillemets

# Structure, pour les deux formats
S001 = {field} : type {expected} attendu, type {found} reçu
S001.point = {field} : un point s'écrit [x, y], deux entiers
S001.segment = {field} : un élément de tracé est un point [x, y] ou une courbe [[x, y], [x, y], [x, y]]
S002 = clé inconnue `{key}` ({in})
S003 = clé obligatoire `{key}` absente ({in})
S004 = format inattendu `{found}` : on attend `{expected}`
S005 = version {found} non prise en charge : les versions 1 à {supported} sont lues
S006 = {field} : valeur inconnue `{value}` ; on attend l'une de {words}
S007.empty = {field} : identifiant vide
S007.control = {field} : l'identifiant contient un caractère de contrôle
S007.long = {field} : identifiant trop long ({length} caractères, {max} au plus)
S008 = {field} : {value} hors de l'intervalle de {min} à {max}

G003 = {subject} : `{key}` désigne le nœud « {node} », qui n'existe pas

G006 = identifiant de port « {id} » déjà employé

G007.unknown = {subject} : `{key}` désigne le port « {port} », qui n'existe pas

G008 = {subject} : `{side}` ne peut pas accompagner `{port}`

G010.side = {subject} : une position exige `side`
G010.border = {subject} : la position ({x}, {y}) n'est pas sur le côté `{side}` de son nœud
G010.container = {subject} : un nœud qui a des enfants ne peut pas positionner ses ports

# Lecteur du format source (.fg)
T001 = octets invalides : le texte doit être en UTF-8
T002 = caractère de contrôle {char} refusé : seuls la tabulation et les fins de ligne sont admis
T003 = chaîne non fermée avant la fin de sa ligne
T004 = échappement inconnu `{escape}`
T005 = l'échappement `{escape}` n'est pas une valeur scalaire Unicode
T006 = `}` sans bloc ouvert
T007 = bloc non fermé avant la fin du fichier
T008.top = hors de tout élément, une ligne est une tête d'élément ou, avant le premier élément, une directive
T008.kind = une tête d'élément commence par un mot de sorte, qui ne commence pas par `@`
T008.id = une tête d'élément donne un identifiant après sa sorte, sous forme de mot
T008.label = seule une étiquette, entre guillemets, peut se placer entre l'identifiant et `{`
T009 = identifiant `{id}` : lettres ASCII, chiffres et `_` seulement, sans chiffre en tête
T010 = directive `@{key}` après le premier élément : l'en-tête précède tous les éléments
T011 = `@` sans clé
T012 = identifiant `{id}` déjà défini en {line}:{col}
T013 = clé `@{key}` déjà donnée en {line}:{col}
T014 = plus de {max} blocs imbriqués
T015 = référence `{id}` : aucun élément ne porte cet identifiant

# La ligne de commande du produit, `foregram`
cli.usage = usage : foregram [--lang <code>] [--json] <commande> ...\n       foregram validate <graphe.json> [<placement.json>...]\n       foregram canon <fichier.json>\n       foregram metrics <graphe.json> <placement.json>...\n       foregram render <graphe.json> <placement.json> <sortie.svg|sortie.png> [--scale <n>]\n       foregram --help | --version\n\n  validate  Valide un graphe, puis chaque placement par rapport à ce graphe.\n  canon     Lit un graphe ou un placement, reconnu à sa clé `format`, et\n            l'écrit sous sa forme canonique, version 2, sur la sortie.\n  metrics   Confronte chaque placement au graphe et écrit ses mesures, une\n            ligne par mesure.\n  render    Confronte le placement au graphe et le dessine, d'aspect neutre,\n            dans un fichier SVG ou PNG choisi par son extension ; --scale\n            donne au PNG ses pixels par pixel du dessin, de 1 à 8.\n\n  Un fichier nommé - est l'entrée standard, une fois par appel.\n  --lang    Langue des messages : `en` (par défaut) ou `fr`.\n  --json    Résultats et refus en JSON, un objet par ligne, tous sur la\n            sortie standard ; voir docs/exchange-format.md.\n\ncodes de sortie : 0 tout est valide, 1 un fichier est refusé, 2 appel\n                 incorrect ou fichier illisible
cli.stdin = <stdin>
cli.lang-unknown = langue inconnue `{lang}` : les catalogues sont `en` et `fr`
cli.valid-graph = {path}: graphe valide ({nodes}, {edges})
